{{- $project := .Get 0 -}} {{- $diskPath := printf "static/latex_files/%s" $project -}} {{- $webPath := printf "latex_files/%s" $project | relURL -}} {{- $files := readDir $diskPath -}} {{- $pdfs := slice -}} {{- range $files -}} {{- if and (not .IsDir) (strings.HasSuffix (lower .Name) ".pdf") -}} {{- $pdfs = $pdfs | append .Name -}} {{- end -}} {{- end -}} {{- $sorted := slice -}} {{- range $pdfs -}} {{- $version := replaceRE "^.*-v([0-9]+)\\.([0-9]+)\\.pdf$" "$1.$2" . -}} {{- $parts := split $version "." -}} {{- $major := int (index $parts 0) -}} {{- $minor := int (index $parts 1) -}} {{- $key := printf "%010d.%010d" $major $minor -}} {{- $sorted = $sorted | append (dict "name" . "key" $key) -}} {{- end -}} {{- $sorted = sort $sorted "key" "desc" -}}
{{- range $sorted }} {{ .name }} {{- end }}
